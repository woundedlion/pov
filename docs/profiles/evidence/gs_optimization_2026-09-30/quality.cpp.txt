#include "core/engine/engine.h"
#include "effects/GSReactionDiffusion.h"
#include "tests/test_fixture.h"
#include <algorithm>
#include <cmath>
#include <cstdio>
#include <vector>

namespace hs_test::effects_tests {
struct GSWhiteBox {
  using GS = GSReactionDiffusion<288, 144>;
  using Grid = Scan::Shader::SsaaGrid<288, 144>;
  static Pixel reference(const GS &gs, int seed, const math::Vector &rv,
                         const math::Vector *nodes, const Grid &grid, int x) {
    if (seed < 0) return Pixel(0, 0, 0);
    int center = gs.refine_render_center(rv, nodes, seed);
    math::Vector positions[7];
    uint16_t concentrations[7];
    positions[0] = nodes[center];
    concentrations[0] = gs.state.B[center];
    for (int j = 0; j < 6; ++j) {
      int ni = ReactionGraph::neighbors[center][j];
      positions[j + 1] = nodes[ni];
      concentrations[j + 1] = gs.state.B[ni];
    }
    uint32_t r = 0, g = 0, b = 0;
    for (int i = 0; i < 4; ++i) {
      math::Vector v = grid.at(x, i);
      float tw = 0, wb = 0;
      for (int j = 0; j < 7; ++j) {
        float dx = v.x - positions[j].x;
        float dy = v.y - positions[j].y;
        float dz = v.z - positions[j].z;
        float d2 = dx * dx + dy * dy + dz * dz;
        float u = 1.0f - d2 * GS::INV_R2;
        if (u > 0) {
          float w = u * u;
          wb += concentrations[j] * w;
          tw += w;
        }
      }
      if (tw <= GS::KERNEL_MIN_TOTAL_WEIGHT) continue;
      float concentration = wb * (GS::Q16_INV / tw);
      if (concentration < GS::B_CULL_THRESHOLD) continue;
      float t = hs::clamp((concentration - GS::B_COLOR_FLOOR) * GS::B_COLOR_SCALE, 0.0f, 1.0f);
      Pixel sample = gs.palette.get_color_unit(t);
      r += (static_cast<uint32_t>(sample.r) + 2u) / 4u;
      g += (static_cast<uint32_t>(sample.g) + 2u) / 4u;
      b += (static_cast<uint32_t>(sample.b) + 2u) / 4u;
    }
    return Pixel(static_cast<uint16_t>(std::min(r, 65535u)),
                 static_cast<uint16_t>(std::min(g, 65535u)),
                 static_cast<uint16_t>(std::min(b, 65535u)));
  }

  static void compare(GS &gs, const char *label, int frame) {
    ScratchScope scope(scratch_arena_a);
    auto lattice = gs.orient_lattice();
    const auto *nodes = lattice.get();
    Grid grid;
    math::TrigLUT<288, 144>::init();
    uint64_t absolute = 0, squared = 0;
    int changed = 0, coverage = 0, lit = 0, hard = 0, maximum = 0;
    double sx = 0, sy = 0, sxx = 0, syy = 0, sxy = 0;
    for (int y = 0; y < 144; ++y) {
      grid.set_row(y);
      for (int x = 0; x < 288; ++x) {
        Fragment frag;
        frag.pos = math::pixel_to_vector<288, 144>(x, y);
        gs.seed_face_lut(frag);
        int seed = static_cast<int>(frag.v0);
        Pixel expected = reference(gs, seed, frag.pos, nodes, grid, x);
        Pixel actual = gs.shade_pixel(seed, frag.pos, nodes, grid, x);
        const int aa[3] = {actual.r, actual.g, actual.b};
        const int ee[3] = {expected.r, expected.g, expected.b};
        int largest = 0;
        for (int c = 0; c < 3; ++c) {
          int error = std::abs(aa[c] - ee[c]);
          absolute += error;
          squared += static_cast<uint64_t>(error) * error;
          largest = std::max(largest, error);
          sx += ee[c]; sy += aa[c]; sxx += double(ee[c]) * ee[c];
          syy += double(aa[c]) * aa[c]; sxy += double(aa[c]) * ee[c];
        }
        maximum = std::max(maximum, largest);
        changed += largest != 0;
        hard += largest > 4096;
        bool al = actual.r || actual.g || actual.b;
        bool el = expected.r || expected.g || expected.b;
        coverage += al != el;
        lit += al || el;
      }
    }
    constexpr double COUNT = 288.0 * 144.0 * 3.0;
    double mx = sx / COUNT, my = sy / COUNT;
    double vx = sxx / COUNT - mx * mx, vy = syy / COUNT - my * my;
    double cov = sxy / COUNT - mx * my;
    constexpr double C1 = 655.35 * 655.35, C2 = 1966.05 * 1966.05;
    double ssim = ((2 * mx * my + C1) * (2 * cov + C2)) /
                  ((mx * mx + my * my + C1) * (vx + vy + C2));
    std::printf("%s,%d,%d,%d,%d,%d,%d,%.9f,%.9f,%.12f,%d,%d\n", label, frame,
                changed, coverage, lit, hard, maximum, absolute / COUNT,
                std::sqrt(squared / COUNT), ssim,
                gs.transition.grow_frames, gs.transition.dissolve_frames);
    std::fflush(stdout);
  }

  static void synthetic(GS &gs, int field) {
    for (int i = 0; i < GS::RD_N; ++i) {
      auto p = ReactionGraph::node(i);
      float b = field == 0 ? 0.1f : field == 1 ? 0.15f + 0.1f * p.y :
                field == 2 ? (p.x > 0 ? 0.7f : 0.0f) :
                (0.15f + 0.14f * std::sin(p.x * 35.0f) * std::cos(p.z * 25.0f));
      gs.state.B[i] = gs.to_q16(b);
    }
    compare(gs, "synthetic", field);
  }
};
}

int main() {
  hs_test::reset_globals();
  hs::set_mock_time(0, 0);
  hs_test::effects_tests::GSWhiteBox::GS gs;
  gs.init();
  std::puts("field,frame,changed_pixels,coverage_changed,lit,hard,max_channel,mae,rmse,global_ssim,grow_frame,dissolve_frame");
  constexpr int FRAMES[] = {1, 4, 16, 40, 160, 320, 490, 700, 1000, 1600, 2000};
  int selected = 0;
  for (int frame = 1; frame <= 2000; ++frame) {
    hs::set_mock_time(static_cast<unsigned long>(frame * 62500 / 1000),
                      static_cast<unsigned long>(frame * 62500));
    gs.draw_frame();
    gs.advance_display();
    if (frame == FRAMES[selected]) {
      hs_test::effects_tests::GSWhiteBox::compare(gs, "evolved", frame);
      ++selected;
    }
  }
  for (int field = 0; field < 4; ++field)
    hs_test::effects_tests::GSWhiteBox::synthetic(gs, field);
}
